## Interrupción

Primero debería declarar una interrupción tal que se ejecute malloco si se desea allocar memoria y chau si se desea liberar memora de alguna dirección.

``` asm
; Syscall para malloco
_isr100:
  pushad
  ; Aca tendría que pasarle como parámetro (o sea, por la pila) el size
  ; Puedo usar la convención que quiera para pasar el parámetro size a la interrupción? Asumo que si
  ; Pusheo la cantidad de memoria que se pide
  push eax
  call malloco
  
  add esp, 4
  ; Modifico en la posición del eax el resultado que devolvió malloco
  mov [esp + offset_EAX], eax

  popad
  iret
```

``` asm
; Syscall para chau
_isr101:
  pushad

  ; Pusheo la direccion virtual de memoria que se quiere liberar
  push eax
  call chau
  
  add esp, 4

  popad
  iret
```
``` asm
_isr14:
  ; Estamos en un page fault.
  pushad    

    ; En el cr2 se encuentra la dirección lineal que generó el page fault
    mov eax, cr2
    push eax
    
    call page_fault_handler

    ; Restauro la pila
    add esp, 4

    ; Si es true se atendio la excepcion
    cmp al, 1
    je .fin

  .ring0_exception:
	  ; Si llegamos hasta aca es que cometimos un page fault fuera del area compartida.
    call kernel_exception
    jmp $

  .fin:
	  popad
	  add esp, 4 ; error code
	  iret

```

Pensando en que el sistema de asignación de memoria es _lazy allocation_ pienso en que tendría que estar en la rutina de atención del page fault esta funcionalidad de alocar el espacio que se pide cuando se quiera acceder a la memoria que se le dijo que tiene

``` C
RESERVABLE_START = 0xA10C0000;
// 0x400000 = PAGE_SIZE * 1024
RESERVABLE_END = 0xA10C0000 + 0x400000;

bool page_fault_handler(vaddr_t virt) {
  uint32_t cr3;
  
  print("Atendiendo page fault...", 0, 0, C_FG_WHITE | C_BG_BLACK);

  // Chequeemos si el acceso fue dentro del area de memoria virtual reservable
  if(!((RESERVABLE_START <= virt) && (virt <= RESERVABLE_END))){
    return false;
  }

  // En caso de que si, mapear la pagina
  cr3 = rcr3();
  paddr_t phy = mmu_next_free_user_page();
  mmu_map_page(cr3, virt, phy, MMU_P | MMU_W | MMU_U);
  
  // Limpio la pantalla
  screen_draw_box(0,0,1,VIDEO_COLS,' ', C_BG_BLACK);
  
  return true;  
}
```
