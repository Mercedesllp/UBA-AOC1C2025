``` C
#define PAGE_SIZE 4096
#define TOPE 1024
#define RESERVABLE_START 0xA10C0000

void* malloco(size_t size){
  uin32_t cant_paginas = ceil(size / PAGE_SIZE);

  for(int i = 0; i < MAX_SLOTS; i++){
    void* virt;

    if((tasks_alocada[current_task][i] != 0) && (i != 0)){
      // Uso la primera posición para indicar que cantidad de páginas ya fueron alocadas
      if((tasks_alocada[0] + cant_paginas) > TOPE){
        virt = NULL;
        break;
      }

      // Sumo al contador de páginas utilizadas
      tasks_alocada[0] += cant_paginas;

      alocada_entry_t entry;
      // Obtengo la dirección a la que le puedo asignar en un futuro la página
      entry.virt, virt = RESERVABLE_START + (tasks_alocada[0] * PAGE_SIZE);
      // Está sin mapear

      // El tamaño lo mido en la cantidad de páginas que ocupa
      entry.tamaño = cant_paginas;
      // Lo guardo en mi array
      tasks_alocada[current_task][i] = entry;
      break;
    }
  }
  
  return virt;
}
```

``` C
// mmu.c

PAGE_SIZE = 4096
RESERVABLE_START = 0xA10C0000;
// 0x400000 = PAGE_SIZE * 1024
// 4 MB luego:
RESERVABLE_END = 0xA10C0000 + 0x400000;

bool page_fault_handler(vaddr_t virt) {
  uint32_t cr3;
  
  print("Atendiendo page fault...", 0, 0, C_FG_WHITE | C_BG_BLACK);

  // Chequeemos si el acceso fue dentro del area de memoria virtual reservable
  if(!((RESERVABLE_END <= virt) && (virt <= RESERVABLE_START))){
    return false;
  }

  // Chequeamos que haya sido pedida la memoria anteriormente (se hizo malloco previamente)
  // (RESERVABLE_START + tasks_alocada[current_task][0] * PAGE_SIZE) == Posicion limite que se pidio memoria
  if((RESERVABLE_START + tasks_alocada[current_task][0] * PAGE_SIZE) < virt){
    return false;
  }
  
  // En caso de que si, mapear la pagina que se busca
  cr3 = rcr3();
  paddr_t phy = mmu_next_free_user_page();
  // Vacío los contenidos que tenía phy
  zero_page(phy);
  mmu_map_page(cr3, virt, phy, MMU_P | MMU_W | MMU_U);
  
  // Limpio la pantalla
  screen_draw_box(0,0,1,VIDEO_COLS,' ', C_BG_BLACK);
  
  return true;  
}
```
